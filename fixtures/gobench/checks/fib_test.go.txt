package fib

import "testing"

// Copied into place by the go-fib-memo gate. Fib(90) under a 30 s timeout fails a version
// without memoization: the naive recursion makes about 10^18 calls.
func TestHiddenFib(t *testing.T) {
	cases := []struct{ n, want int }{
		{0, 0}, {1, 1}, {2, 1}, {3, 2}, {10, 55}, {20, 6765},
		{50, 12586269025}, {90, 2880067194370816120},
	}
	for _, c := range cases {
		if got := Fib(c.n); got != c.want {
			t.Errorf("Fib(%d) = %d, want %d", c.n, got, c.want)
		}
	}
}
