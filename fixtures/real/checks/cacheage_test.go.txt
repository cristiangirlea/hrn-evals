package cacheage_test

import (
	"regexp"
	"strings"
	"testing"
	"time"

	"github.com/cristiangirlea/hrn-evals/fixtures/real/cacheage"
)

// negative matches a negative duration in a reason, such as "idle -5ms".
var negative = regexp.MustCompile(`-[0-9]`)

func TestDecide(t *testing.T) {
	now := time.Date(2026, 10, 16, 12, 0, 0, 0, time.UTC)
	for _, tc := range []struct {
		name    string
		written time.Time
		idle    time.Duration
		remove  bool
	}{
		{"an hour idle, limit 30m", now.Add(-time.Hour), 30 * time.Minute, true},
		{"a minute idle, limit 30m", now.Add(-time.Minute), 30 * time.Minute, false},
		{"written now, --idle 0", now, 0, true},
		// File times and the clock differ by their resolution.
		{"written a hair after now, --idle 0", now.Add(5 * time.Millisecond), 0, true},
		{"written after now, limit 1ns", now.Add(time.Second), time.Nanosecond, false},
	} {
		remove, why := cacheage.Decide(cacheage.Cache{Path: "c", LastBuild: tc.written}, now, tc.idle)
		if remove != tc.remove {
			t.Errorf("%s: remove = %v (%s), want %v", tc.name, remove, why, tc.remove)
		}
		if negative.MatchString(why) {
			t.Errorf("%s: a negative age reached the reason: %q", tc.name, why)
		}
	}
}

func TestCandidates(t *testing.T) {
	now := time.Date(2026, 10, 16, 12, 0, 0, 0, time.UTC)
	caches := []cacheage.Cache{
		{Path: "fresh", LastBuild: now.Add(3 * time.Millisecond)},
		{Path: "old", LastBuild: now.Add(-48 * time.Hour)},
		{Path: "day", LastBuild: now.Add(-24 * time.Hour)},
	}
	var got []string
	for _, c := range cacheage.Candidates(caches, now, 0) {
		got = append(got, c.Path)
	}
	if strings.Join(got, ",") != "old,day,fresh" {
		t.Errorf("Candidates(min idle 0) = %v, want old, day, fresh", got)
	}
	got = nil
	for _, c := range cacheage.Candidates(caches, now, 30*time.Hour) {
		got = append(got, c.Path)
	}
	if strings.Join(got, ",") != "old" {
		t.Errorf("Candidates(min idle 30h) = %v, want old", got)
	}
}
